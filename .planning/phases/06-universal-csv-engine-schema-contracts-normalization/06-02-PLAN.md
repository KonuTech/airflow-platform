# packages/dataplat/src/dataplat/config/model.py — new additions

class ColumnContract(BaseModel):
    """One column's type, nullability, business-key and semantic contract (SCHEMA-02)."""
    model_config = ConfigDict(extra="forbid", frozen=True)
    name: str
    type: str  # "string" | "integer" | "decimal" | "date" | "timestamp" | "boolean" -- closed set, plain str (D-19/project convention)
    nullable: bool
    required: bool
    business_key: bool = False
    description: str = ""  # D-19: free-text semantics now, controlled tags deferred
    format: str | None = None       # strptime format, for type in {"date", "timestamp"} (CSV-09)
    fixed_width: int | None = None  # declared width for a "string" identifier column (fixture 51)
    reject_scientific_notation: bool = False  # for a "string" identifier column (fixture 50)

class FilenameMaskConfig(BaseModel):
    """A dataset's opt-in filename mask (D-07/D-08/D-10). Absent entirely when a dataset has none."""
    model_config = ConfigDict(extra="forbid", frozen=True)
    mask: str  # e.g. "{dataset}_{country}_{business_date:%Y%m%d}[_{seq:03d}].csv"

class NormalizationConfig(BaseModel):
    """A dataset's opt-in, single locale/normalization profile (D-12/D-13/D-14). Absent when a
    dataset has no numeric/currency/boolean columns needing one (e.g. customers)."""
    model_config = ConfigDict(extra="forbid", frozen=True)
    decimal_separator: str = "."
    thousands_separator: str | None = None
    currency_symbols: list[str] = []
    percent_as_fraction: bool = True
    negative_style: str = "leading-minus"  # "leading-minus" | "trailing-minus" | "parentheses"
    null_tokens: list[str] = [""]  # D-14: empty string only, by default
    null_sentinels: dict[str, list[str]] = {}  # per-column value sentinels, e.g. {"amount": ["-1"]}
    boolean_true_tokens: list[str] = []
    boolean_false_tokens: list[str] = []
    two_digit_year_pivot: int | None = None
    spreadsheet_epoch: str | None = None  # "1900" | "1904"
    timezone: str | None = None  # IANA zone name, for naive local timestamps
    ambiguous_time_policy: str = "reject"  # "reject" | "earliest" | "latest"

class CsvParsingConfig(BaseModel):
    """Structural CSV-parsing contract overrides (CSV-05/07/08). Every field None/default means
    "detect it" -- customers.yaml declares none of this, matching Phase 3's already-correct UTF-8/
    comma/header-row-0 shape once real detection confirms the same answer."""
    model_config = ConfigDict(extra="forbid", frozen=True)
    encoding: str | None = None
    delimiter: str | None = None
    quotechar: str = '"'
    max_field_bytes: int = 1_048_576  # supersedes csv_processor.source.FIELD_SIZE_LIMIT (LOAD-07)
    header_row: int | None = None
    skip_footer_rows: int = 0
    header_trim: bool = False
    header_case_sensitive: bool = True
